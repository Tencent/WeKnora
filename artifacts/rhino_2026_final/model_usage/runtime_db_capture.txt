SQL
---
SELECT model_id, model_name, call_type,
       COUNT(*) AS calls,
       SUM(input_tokens) AS input_tokens,
       COUNT(input_tokens) AS input_observed,
       SUM(output_tokens) AS output_tokens,
       SUM(total_tokens) AS total_tokens,
       ROUND(AVG(latency_ms)::numeric, 6) AS avg_latency_ms,
       COUNT(*) FILTER (WHERE prompt_cache_status='hit') AS prompt_hit_calls,
       COUNT(*) FILTER (WHERE prompt_cache_status='miss') AS prompt_miss_calls,
       COUNT(*) FILTER (WHERE prompt_cache_status IS NULL) AS prompt_null_calls,
       SUM(cache_read_tokens) AS cache_read_tokens,
       SUM(cache_miss_tokens) AS cache_miss_tokens,
       ROUND((COUNT(*) FILTER (WHERE prompt_cache_status='hit'))::numeric /
             NULLIF(COUNT(*) FILTER (WHERE prompt_cache_status IN ('hit','miss')),0), 6)
         AS prompt_call_hit_rate,
       ROUND(SUM(cache_read_tokens)::numeric / NULLIF(SUM(input_tokens),0), 6)
         AS prompt_token_cache_ratio,
       COUNT(*) FILTER (WHERE embedding_cache_status='hit') AS embedding_hit_calls,
       COUNT(*) FILTER (WHERE embedding_cache_status='miss') AS embedding_miss_calls
FROM model_usage
WHERE tenant_id=10000
  AND model_id='5a50bf0a-60f9-4340-974b-0bd85c80b286'
  AND created_at >= '2026-08-31T13:00:00Z'::timestamptz
  AND created_at <  '2026-09-07T11:00:00Z'::timestamptz
GROUP BY model_id, model_name, call_type;
